----------------------------------------------------------------
-- AUTH
----------------------------------------------------------------
local function isAuth(plr)
	if #ALLOWED_USERS == 0 then return true end

	for _, name in ipairs(ALLOWED_USERS) do
		if plr.Name:lower() == name:lower() then
			return true
		end
	end

	warn("[REQUIRE HUB] Not authorized: " .. plr.Name .. " (" .. plr.UserId .. ")")
	return false
end

----------------------------------------------------------------
-- TARGET RESOLVING (Name or DisplayName, exact or unambiguous partial)
----------------------------------------------------------------
local function getTargetUsername(plr, input)
	if typeof(input) ~= "string" then
		return plr.Name
	end

	local username = input:match("^%s*@?(.-)%s*$")
	if username == "" then
		return plr.Name
	end

	local lowered = username:lower()

	for _, player in ipairs(Players:GetPlayers()) do
		if player.Name:lower() == lowered or player.DisplayName:lower() == lowered then
			return player.Name
		end
	end

	local match
	for _, player in ipairs(Players:GetPlayers()) do
		if player.Name:lower():sub(1, #lowered) == lowered
			or player.DisplayName:lower():sub(1, #lowered) == lowered then
			if match and match ~= player.Name then
				warn("[REQUIRE HUB] Ambiguous username: " .. username)
				return username
			end
			match = player.Name
		end
	end

	if not match then
		warn("[REQUIRE HUB] No player found for: " .. username)
	end

	return match or username
end

----------------------------------------------------------------
-- RUNNER
----------------------------------------------------------------
local function runScript(plr, pIdx, sIdx, input)
    if not isAuth(plr) then
        return
    end

    local page = PAGES[pIdx]
    local item = page and page.Scripts[sIdx]

    if not item or type(item[2]) ~= "function" then
        warn("[REQUIRE HUB] Invalid script selection")
        return
    end

    local target = getTargetUsername(plr, input)

    task.spawn(function()
        print(
            "[REQUIRE HUB] Running "
                .. item[1]
                .. " for "
                .. target
        )

        local success, err = pcall(function()
            item[2](target)
        end)

        if success then
            print("[REQUIRE HUB] Callback finished: " .. item[1])
        else
            warn(
                "[REQUIRE HUB] "
                    .. item[1]
                    .. " failed: "
                    .. tostring(err)
            )
        end
    end)
end

----------------------------------------------------------------
-- R6 CONVERTER
----------------------------------------------------------------
local function setR6(targetPlr)
	if not targetPlr or not targetPlr.Parent then return end

	local oldChar = targetPlr.Character
	local cf = oldChar and oldChar:GetPivot() or CFrame.new(0, 10, 0)

	local ok, desc = pcall(function()
		return Players:GetHumanoidDescriptionFromUserId(targetPlr.UserId)
	end)
	if not ok or not desc then
		desc = Instance.new("HumanoidDescription")
	end

	local model = Players:CreateHumanoidModelFromDescription(desc, Enum.HumanoidRigType.R6)
	model.Name = targetPlr.Name

	if oldChar then
		oldChar:Destroy()
	end

	targetPlr.Character = model
	model.Parent = workspace
	model:PivotTo(cf)
end

----------------------------------------------------------------
-- RESET (respawn in place)
----------------------------------------------------------------
local function resetTarget(targetPlr)
	if not targetPlr or not targetPlr.Parent then return end

	local oldChar = targetPlr.Character
	local cf = oldChar and oldChar:GetPivot()

	targetPlr:LoadCharacter()

	local newChar = targetPlr.Character
	if newChar and cf then
		newChar:PivotTo(cf)
	end
end

----------------------------------------------------------------
-- HINT SPAWNER
----------------------------------------------------------------
local function spawnHint(text, duration)
	local h = Instance.new("Hint")
	h.Name = "HubHint"
	h.Text = text
	h.Parent = workspace

	-- duration = nil means infinity (stays until cleared)
	if duration then
		Debris:AddItem(h, duration)
	end
end

local function clearHints()
	local count = 0
	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Hint") and obj.Name == "HubHint" then
			obj:Destroy()
			count += 1
		end
	end
	return count
end

local function ui(cls, props, rad, strokeCol)
    local o = Instance.new(cls)

    for k, v in pairs(props) do
        o[k] = v
    end

    if o:IsA("GuiObject") then
        o.BorderSizePixel = 0
    end

    if o:IsA("TextLabel")
        or o:IsA("TextButton")
        or o:IsA("TextBox") then

        o.Font = Enum.Font.Code
        o.TextStrokeTransparency = 1
    end

    if o:IsA("TextButton") then
        o.AutoButtonColor = false
    end

    -- Intentionally omit UICorner for square edges.
    if strokeCol then
        local stroke = Instance.new("UIStroke")
        stroke.Color = strokeCol
        stroke.Thickness = 1
        stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        stroke.Parent = o

        return o, stroke
    end

    return o
end

	local function go(n, dir)
		if busy then return end

		if n == cur then return end
		busy = true
		list.Visible = false

		local oldF = slides[cur]
		local newF = slides[n]

		newF.Position = UDim2.new(dir,0,0,0)
		newF.Visible = true

		TweenService:Create(oldF, info, {
			Position = UDim2.new(-dir,0,0,0)
		}):Play()

		local tIn = TweenService:Create(newF, info, {
			Position = UDim2.new(0,0,0,0)
		})

		tIn:Play()

		ind.Text = PAGES[n].Category .. "  (" .. n .. " / " .. #PAGES .. ")"
		cur = n

		tIn.Completed:Connect(function()
			oldF.Visible = false
			busy = false
		end)
	end

local addingHint = false

addHintBtn.MouseButton1Click:Connect(function()
    if not isAuth(plr) or addingHint then return end

    -- Same reading pattern as the username input,
    -- but using the separate hint TextBox.
    local input = hintBox.Text


    local text = input:match("^%s*(.-)%s*$")
    local durationInput = durationBox.Text
    local raw = durationInput:match("^%s*(.-)%s*$"):lower()

    local permanent = raw == "infinity"
        or raw == "inf"
        or raw == "∞"

    local duration

    if not permanent then
        duration = tonumber(raw)

        if not duration
            or duration ~= duration
            or duration <= 0
            or duration == math.huge then
            setHintStatus(
                "Enter positive seconds or infinity.",
                true
            )
            return
        end
    end

    addingHint = true
    addHintBtn.Text = "ADDING..."
    setHintStatus("Adding hint...", false)

    task.spawn(function()
        local success, err = pcall(function()
            local result = TextService:FilterStringAsync(
                text,
                plr.UserId
            )

            local filteredText =
                result:GetNonChatStringForBroadcastAsync()

            if not gui.Parent or plr.Parent ~= Players then
                return
            end

            spawnHint(filteredText, duration)
        end)

        addingHint = false

        if not gui.Parent then return end

        addHintBtn.Text = "ADD HINT"

        if success then
            setHintStatus(
                permanent
                    and "Hint added — stays until cleared."
                    or ("Hint added — removal scheduled in "
                        .. tostring(duration) .. " seconds."),
                false
            )
        else
            setHintStatus(
                "Couldn't add hint: " .. tostring(err),
                true
            )
        end
    end)
end)

local executorRunId = 0

local function executorString(value)
    local success, result = pcall(tostring, value)

    if success then
        return result
    end

    return "<unprintable value>"
end

local function executorTraceback(err)
    local message = executorString(err)

    local success, traceback = pcall(function()
        return debug.traceback(message, 2)
    end)

    return success and traceback or message
end

local function executorPreview(message, limit)
    local compact = message:gsub("[\r\n]+", " ")

    if #compact > limit then
        return compact:sub(1, limit) .. "..."
    end

    return compact
end

executeBtn.MouseButton1Click:Connect(function()
    if not isAuth(plr) then
        return
    end

    if executorRunning then
        setExecutorStatus(
            "A script is running. Check Output for details.",
            true
        )
        return
    end

    local source = sourceBox.Text

    if typeof(source) ~= "string" or not source:find("%S") then
        setExecutorStatus("Enter code or select a preset.", true)
        return
    end

    source = source:gsub("^\239\187\191", "")
    source = source:gsub("\r\n", "\n"):gsub("\r", "\n")

    if typeof(loadstring) ~= "function" then
        setExecutorStatus(
            "loadstring is unavailable in this environment.",
            true
        )
        return
    end

    executorRunning = true
    executorRunId += 1

    local runId = executorRunId
    local chunkName = "RequireHub_" .. plr.UserId .. "_" .. runId
    local startedAt = os.clock()

    executeBtn.Text = "COMPILING..."
    setExecutorStatus("Compiling script...", false)

    local function finish(message, isError)
        executorRunning = false

        if executeBtn.Parent then
            executeBtn.Text = "EXECUTE"
        end

        setExecutorStatus(message, isError)
    end

    local compileSuccess, compiled, compileError = pcall(function()
        return loadstring(source, "=" .. chunkName)
    end)

    if not compileSuccess then
        local message = executorString(compiled)

        warn("[REQUIRE HUB] Compiler failure:\n" .. message)

        finish(
            "Compiler failure: " .. executorPreview(message, 160),
            true
        )
        return
    end

    if typeof(compiled) ~= "function" then
        local message = executorString(
            compileError or "Compiler did not return a function."
        )

        warn("[REQUIRE HUB] Syntax error:\n" .. message)

        finish(
            "Syntax error: " .. executorPreview(message, 160),
            true
        )
        return
    end

    executeBtn.Text = "RUNNING..."
    setExecutorStatus("Running script #" .. runId .. "...", false)

    print("[REQUIRE HUB] Starting " .. chunkName)

    task.spawn(function()
        local results = table.pack(
            xpcall(compiled, executorTraceback)
        )

        local elapsed = os.clock() - startedAt

        if not results[1] then
            local message = executorString(results[2])

            warn(
                "[REQUIRE HUB] "
                    .. chunkName
                    .. " failed:\n"
                    .. message
            )

            finish(
                "Runtime error: "
                    .. executorPreview(message, 150)
                    .. " | See Output.",
                true
            )
            return
        end

        local returned = {}

        for index = 2, results.n do
            returned[#returned + 1] = executorString(results[index])
        end

        local message = string.format(
            "Finished in %.2fs.",
            elapsed
        )

        if #returned > 0 then
            local values = table.concat(returned, ", ")

            print("[REQUIRE HUB] Returned: " .. values)

            message ..= " Returned: " .. executorPreview(values, 100)
        end

        print(
            "[REQUIRE HUB] "
                .. chunkName
                .. " finished in "
                .. string.format("%.2f", elapsed)
                .. "s"
        )

        finish(message, false)
    end)
end)

local function resizeHub()
    local expanded = currentTab == "Executor"
    local width = expanded and executorWidth or normalWidth
    local height = minimized and 46
        or (expanded and executorHeight or normalHeight)

    local oldSize = main.Size
    local deltaWidth = width - oldSize.X.Offset
    local deltaHeight = height - oldSize.Y.Offset

    main.Position = UDim2.new(
        main.Position.X.Scale,
        main.Position.X.Offset - deltaWidth / 2,
        main.Position.Y.Scale,
        main.Position.Y.Offset - deltaHeight / 2
    )

    main.Size = UDim2.new(0, width, 0, height)
end

local function showTab(name)
    currentTab = name
    list.Visible = false

    scriptsTab.Visible = name == "Scripts" and not minimized
    general.Visible = name == "Hints" and not minimized
    executorTab.Visible = name == "Executor" and not minimized
    creditsTab.Visible = name == "Credits" and not minimized

    for tabName, button in pairs(tabBtns) do
        local active = tabName == name
        button.BackgroundColor3 = active and C.Cyan or C.Card
        button.TextColor3 = active and C.Head or C.Cyan
    end

    resizeHub()
end

for name, button in pairs(tabBtns) do
    button.MouseButton1Click:Connect(function()
        showTab(name)
    end)
end

showTab("Scripts")
